## The checks

Run under `python3 -I` on 2026-11-20 with numpy. The first script checks
the exit time, the deterministic crossing and the queue's mean wait; the
second the contracted bias under an exponential wait, the independence of
retention from a bias, and the overshoot constant. The diffusive drift
before coding is E[σ²W] from the mean wait and was not simulated apart.

    # drift.py
    import numpy as np
    rng = np.random.default_rng(1)
    # 1. exit time of Gaussian random walk from (-d, d): approx d^2/sig^2
    sig = 0.1
    for d in (0.5, 1.0):
        M = 20000; x = np.zeros(M); t = np.zeros(M); alive = np.ones(M, bool); n = 0
        while alive.any():
            n += 1; x[alive] += sig*rng.normal(size=alive.sum()); out = alive & (np.abs(x) >= d); t[out] = n; alive &= ~out
        print(f"exit d={d}: mean {t.mean():.1f} vs d^2/sig^2 {d*d/sig**2:.1f}")
    # 2. contracting toward c at distance D: deterministic crossing of delta
    a, D = 0.9, 2.0
    for delta in (0.5, 1.5, 2.5):
        n = np.log(D/(D-delta))/-np.log(a) if delta < D else float('inf')
        print(f"contract a={a} D={D} delta={delta}: crossing n* = {n:.2f}")
    # 3. M/M/1 mean wait
    kap = 1.0
    for lam in (0.5, 0.8, 0.9, 0.95):
        N = 400000
        arr = np.cumsum(rng.exponential(1/lam, N)); svc = rng.exponential(1/kap, N)
        dep = np.empty(N); prev = 0.0
        for i in range(N):
            prev = max(prev, arr[i]) + svc[i]; dep[i] = prev
        W = (dep - arr)[N//10:]
        print(f"rho={lam/kap:.2f}: E[W] {W.mean():.2f} vs 1/(kap-lam) {1/(kap-lam):.2f}")

    # drift2.py
    import numpy as np
    from math import log, sqrt, pi
    rng = np.random.default_rng(2)
    a, D = 0.9, 2.0
    for r in (0.5, 0.1, 0.05):          # r = kap - lam
        W = rng.exponential(1/r, 1_000_000)
        emp = (D*(1-a**W)).mean(); th = D*log(1/a)/(r+log(1/a))
        print(f"kap-lam={r}: E[bias] emp {emp:.4f} th {th:.4f}")
    s, sig, n, M = 1.0, 0.1, 50, 200000
    x0 = rng.normal(0, s, M)
    for mu in (0.0, 0.5):
        x = x0 + n*mu + sig*np.sqrt(n)*rng.normal(size=M)
        r = np.corrcoef(x0, x)[0,1]; print(f"mu={mu}: I emp {-0.5*np.log(1-r*r):.4f} th {0.5*np.log1p(s*s/(n*sig*sig)):.4f}; mean drift {np.mean(x-x0):.2f}")
    print("overshoot const", -(-1.4603545088)/sqrt(2*pi))
