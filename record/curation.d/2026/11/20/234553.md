## The check behind CLAIM-tmpmugkp

Run under `python3 -I` on 2026-11-20 with numpy:

    # Two-layer linear net W2 W1 learning Sigma = U S V^T from small init (whitened inputs).
    # Compare gradient descent with orthogonalized (spectral-norm steepest descent / duality-map) updates.
    import numpy as np
    rng = np.random.default_rng(0)
    d, h = 8, 8
    s = np.array([8., 4., 2., 1.] + [0.]*4)
    U, _ = np.linalg.qr(rng.normal(size=(d, d))); V, _ = np.linalg.qr(rng.normal(size=(d, d)))
    Sig = U @ np.diag(s) @ V.T
    def polar(G):
        u, _, vt = np.linalg.svd(G); return u @ vt
    def run(kind, lr, steps):
        W1 = 1e-3*rng.normal(size=(h, d)); W2 = 1e-3*rng.normal(size=(d, h))
        hist = []
        for t in range(steps):
            E = Sig - W2 @ W1                      # loss = 1/2 ||Sig - W2 W1||^2
            G2 = -E @ W1.T; G1 = -W2.T @ E
            if kind == "gd": W2 -= lr*G2; W1 -= lr*G1
            else:            W2 -= lr*polar(G2); W1 -= lr*polar(G1)
            hist.append(np.diag(U.T @ (W2 @ W1) @ V)[:4] / s[:4])   # fraction of each mode learned
        hist = np.array(hist)
        half = [int(np.argmax(hist[:, k] >= 0.5)) if (hist[:, k] >= 0.5).any() else None for k in range(4)]
        return half, hist[-1]
    for kind, lr, steps in (("gd", 0.01, 4000), ("orth", 0.01, 4000), ("orth", 0.003, 12000)):
        half, final = run(kind, lr, steps)
        print(f"{kind} lr={lr}: steps to half-learn modes s=8,4,2,1: {half}; final fractions {np.round(final, 2)}")

Output: gradient descent, steps to half-learn s = 8, 4, 2, 1: 95, 179, 345,
617; polar updates at lr = 0.01: 200, 141, 100, 70; at lr = 0.003: 666, 470,
333, 235. √(s/2)/lr gives 200, 141, 100, 71 at lr = 0.01.
