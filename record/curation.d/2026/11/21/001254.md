## The assistant's checks

Run under `python3 -I` on 2026-11-21 with numpy. powerlaw.py fits the
return-probability exponent on designed rate trees; logmetric.py checks that
the four-point δ of the logarithm of a metric stays bounded.

    # powerlaw.py
    # Return probability on a b-ary ultrametric tree with leaf-to-leaf rate g^(level-1):
    # does it decay as a power law, with exponent ln b / ln(1/(b g))?
    import numpy as np, itertools
    def run(b, L, g):
        n = b**L
        leaf = np.array(list(itertools.product(range(b), repeat=L)))
        diff = leaf[:, None, :] != leaf[None, :, :]
        first = np.where(diff.any(-1), diff.argmax(-1), L)      # first differing digit
        lev = L - first                                          # 0 on diagonal, 1 siblings ... L root
        Q = np.where(lev > 0, g**(lev - 1.0), 0.0)
        G = Q - np.diag(Q.sum(1))
        w, V = np.linalg.eigh(G)
        ts = np.logspace(0, np.log10((1/(b*g))**(L-1)), 40)
        P = np.array([np.sum(V[0]**2 * np.exp(w*t)) for t in ts])   # P(at start at time t), start = leaf 0
        m = (ts > 3) & (P - 1/n > 5/n)                           # fit before equilibrium
        slope = np.polyfit(np.log(ts[m]), np.log(P[m] - 1/n), 1)[0]
        return slope, -np.log(b)/np.log(1/(b*g))
    for b, L, g in ((2, 10, 0.1), (2, 10, 0.05), (2, 10, 0.2), (3, 6, 0.1), (4, 5, 0.05)):
        s, pred = run(b, L, g)
        print(f"b={b} L={L} g={g}: fitted exponent {s:.3f}, ln b / ln(1/(b g)) predicts {pred:.3f}")

    # logmetric.py
    # Check: is the four-point delta of log d bounded for any metric d? (assistant's own check of the reader's observation)
    import numpy as np, itertools
    rng = np.random.default_rng(11)
    def delta4(D):
        n = len(D); best = 0.0
        for a, b, c, d in itertools.combinations(range(n), 4):
            s = sorted([D[a,b]+D[c,d], D[a,c]+D[b,d], D[a,d]+D[b,c]]); best = max(best, (s[2]-s[1])/2)
        return best
    worst = 0.0
    for trial in range(400):
        n = 9
        kind = trial % 3
        if kind == 0:   # Euclidean points in random dimension
            X = rng.normal(size=(n, rng.integers(1, 6))) * np.exp(rng.normal(size=(n, 1)) * 2)
            D = np.sqrt(((X[:, None] - X[None]) ** 2).sum(-1))
        else:           # shortest-path metric of a random complete graph with heavy-tailed weights
            W = np.exp(rng.normal(size=(n, n)) * (3 if kind == 1 else 0.3)); W = np.triu(W, 1); W = W + W.T
            D = W.copy()
            for k in range(n): D = np.minimum(D, D[:, [k]] + D[[k], :])
        np.fill_diagonal(D, 1)
        L = np.log(D)
        worst = max(worst, delta4(L))
    print(f"max four-point delta of log d over 400 random 9-point metrics: {worst:.3f}  (ln 2 = {np.log(2):.3f})")
