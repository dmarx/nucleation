## The checks

Run under `python3 -I` on 2026-11-20 with numpy. hyp.py defines the
helpers (commute times by linear solve, the four-point δ, the defect from a
given tree) and runs the designed hierarchy without noise and two flat
chains; hyp2.py adds within-level heterogeneity; hyp3.py checks convergence
to the minimax ultrametric on a torus with random barriers; hyp4.py checks
κ = n·R_eff and the proven interval; hyp5.py sweeps barriers and checks
that the tree changes only at crossings.

    # hyp.py
    import numpy as np, itertools
    def commute(Q):
        # Q: symmetric rate matrix (off-diag rates), CTMC; hitting times by linear solve
        n = len(Q); G = Q - np.diag(Q.sum(1)); H = np.zeros((n, n))
        for j in range(n):
            idx = [i for i in range(n) if i != j]
            h = np.linalg.solve(-G[np.ix_(idx, idx)], np.ones(n-1)); H[idx, j] = h
        return H + H.T
    def delta4(D):
        n = len(D); best = 0.0
        for a, b, c, d in itertools.combinations(range(n), 4):
            s = sorted([D[a,b]+D[c,d], D[a,c]+D[b,d], D[a,d]+D[b,c]])
            best = max(best, (s[2]-s[1])/2)
        return best
    def ultra_defect(D, lev, levels):
        # max deviation of D from best constant per level (ultrametric with given tree)
        c = 0.0
        for k in levels:
            v = D[lev == k]; c = max(c, (v.max()-v.min())/2)
        return c
    L, b = 4, 2; n = b**L
    leaf = np.array(list(itertools.product(range(b), repeat=L)))
    lev = np.zeros((n, n), int)
    for i in range(n):
        for j in range(n):
            if i != j: lev[i, j] = L - np.argmax(leaf[i] != leaf[j])   # 1 = siblings, L = root split
    for g in (0.5, 0.2, 0.1, 0.01, 0.001):
        Q = np.where(lev > 0, g**(lev-1), 0.0)
        D = np.log(commute(Q)); off = lev > 0
        spacing = min(abs(np.mean(D[lev==k+1]) - np.mean(D[lev==k])) for k in range(1, L))
        print(f"hier g={g}: delta={delta4(D):.4f} ultra-defect c={ultra_defect(D, lev, range(1,L+1)):.4f} level spacing={spacing:.3f} diam={D[off].max()-D[off].min():.3f}")
    # non-hierarchical comparisons, same n: cycle and 4x4 torus
    cyc = np.zeros((n, n))
    for i in range(n): cyc[i, (i+1)%n] = cyc[(i+1)%n, i] = 1
    tor = np.zeros((n, n)); m = 4

        for y in range(m):
            i = x*m+y
            for j in (((x+1)%m)*m+y, x*m+(y+1)%m): tor[i, j] = tor[j, i] = 1
    for name, Q in (("cycle", cyc), ("torus", tor)):
        D = np.log(commute(Q)); off = ~np.eye(n, dtype=bool)
        vals = np.unique(np.round(D[off], 6))
        print(f"{name}: delta={delta4(D):.4f} diam={vals.max()-vals.min():.3f} distinct values={len(vals)} min gap between values={np.diff(vals).min():.4f}")

    # hyp3.py
    import numpy as np, itertools, warnings
    warnings.filterwarnings("ignore")
    exec(open("hyp.py").read().split("L, b = 4, 2")[0])
    def minimax(U, A):
        # minimax path barrier over edges in adjacency A (Floyd-Warshall in (min,max))
        n = len(U); M = np.where(A, U, np.inf); np.fill_diagonal(M, 0)
        for k in range(n): M = np.minimum(M, np.maximum(M[:, [k]], M[[k], :]))
        return M
    n, m = 16, 4
    A = np.zeros((n, n), bool)
    for x in range(m):
        for y in range(m):
            i = x*m+y
            for j in (((x+1)%m)*m+y, x*m+(y+1)%m): A[i, j] = A[j, i] = True
    rng = np.random.default_rng(5)
    U = rng.uniform(0, 1, (n, n)); U = np.triu(U, 1); U = U + U.T
    u = minimax(U, A); off = ~np.eye(n, dtype=bool)
    print("minimax barrier is ultrametric: delta(u) =", round(delta4(u), 6))
    for eps in (0.5, 1e-1, 1e-2, 1e-4, 1e-6):

    def clusters(u):
        # the set of classes over all budgets: the tree's topology
        n = len(u); out = set()
        for t in np.unique(u):
            for i in range(n): out.add(frozenset(np.flatnonzero(u[i] <= t)))
        return frozenset(out)
    rng = np.random.default_rng(7)
    U = rng.uniform(0, 1, (n, n)); U = np.triu(U, 1); U = U + U.T
    edges = [(i, j) for i in range(n) for j in range(i+1, n) if A[i, j]]
    vals = sorted(U[i, j] for i, j in edges)
    bad = 0; changes = 0; trials = 0
    for (i, j) in edges[:12]:
        prev = None
        for x in np.linspace(0.001, 0.999, 400):
            V = U.copy(); V[i, j] = V[j, i] = x
            top = clusters(minimax(V, A)); trials += 1
            if prev is not None and top != prev:
                changes += 1
                # a change must straddle another edge's barrier value
                others = [U[a, b] for a, b in edges if (a, b) != (i, j)]
                if not any(px < o <= x for o in others): bad += 1
            prev, px = top, x
    print(f"topology changes {changes} over {trials} settings; changes not at a crossing with another barrier: {bad}")
